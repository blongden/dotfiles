# Debian testing (trixie/forky-sid). Packages for the sway/Wayland desktop.
# Install:  sudo apt update && sudo apt install --no-install-recommends $(grep -vE '^\s*#|^\s*$' packages.txt)
# then deploy configs with stow (see README.md).

# --- deploy tooling ---
stow

# --- display manager / greeter ---
greetd               # minimal Wayland login manager (replaces lightdm)
tuigreet             # console greeter for greetd — fallback (see config.toml header)
cage                 # kiosk compositor (unused now sway hosts ReGreet, kept as alt host)
cargo                # build-only: ReGreet has no prebuilt binary
libgtk-4-dev         # build-only: ReGreet
libadwaita-1-dev     # build-only: ReGreet

# --- compositor + core session ---
sway
swaybg               # wallpaper (output * bg)
swayidle             # idle -> screensaver / dpms
xwayland             # X11 client support
autotiling           # dynamic split direction (python3-i3ipc dep)
python3-i3ipc        # used by autotiling + ~/.config/sway/winswitch.sh
kanshi               # output profiles (replaces autorandr)

# --- bar / launcher / notifications ---
waybar
wofi                 # launcher (drun/run/dmenu)
rofimoji             # emoji / symbol picker via wofi ($mod+. type, $mod+Shift+. copy)
wmenu                # sway default $menu fallback
dunst                # notifications (also herdr's "system" toast target)
adwaita-icon-theme-legacy # fullcolor icons Adwaita's index inherits (dunst recursive icon lookup)
fonts-font-awesome   # waybar glyph fallback

# --- screenshots / clipboard ---
grim
slurp
swappy               # annotate (screenshot.sh edit modes)
wl-clipboard         # wl-copy / wl-paste
cliphist             # clipboard history ($mod+p)

# --- portals (screen-share, file pickers) ---
xdg-desktop-portal
xdg-desktop-portal-gtk
xdg-desktop-portal-wlr

# --- hardware controls used by keybinds / waybar ---
brightnessctl        # backlight keys + waybar module
blueman              # blueman-manager GUI for the waybar bluetooth module (--no-install-recommends)
pipewire-pulse       # provides pactl (volume keys, waybar)
pulseaudio-utils     # pactl if not pulled by pipewire-pulse
network-manager-gnome # nm-applet (tray)
policykit-1-gnome    # polkit auth agent

# --- phone integration ---
kdeconnect           # clipboard / notifications / file share with the phone (autostart.sh)

# --- mail ---
protonmail-bridge     # local IMAP/SMTP for Proton Mail (autostart.sh; re-login once via `protonmail-bridge --cli`)

# --- files / images ---
mc                   # midnight commander ($mod+Shift+m; twin-pane option)
libheif-examples     # heif-convert  (HEIC -> JPG)
heif-gdk-pixbuf      # HEIC support for gdk-pixbuf apps (image viewers, file dialogs)
imagemagick          # `magick` — used by heic2img (HEIC -> png/jpg, incl. to clipboard)
swayimg              # Wayland image viewer (default for HEIC/HEIF; heif via libheif)
ffmpegthumbnailer    # yazi: video thumbnails
fd-find              # yazi: file finding (symlink ~/.local/bin/fd -> fdfind)
fzf                  # yazi: fuzzy jump
zoxide               # yazi: frecency jump

# --- voice to text (push-to-talk dictation, $mod+grave) ---
whisper.cpp          # whisper-cli local transcription (models: ~/.local/share/whisper/ggml-*.bin)
wtype                # type transcribed text into the focused window (Wayland)
wayland-utils        # wayland-info (protocol inspection)

# --- terminal ---
kitty

# --- browser ---
chromium             # default browser + ws2/ws3 --app windows

# --- screensaver engine ---
pipx                 # then: pipx install terminaltexteffects   (provides `tte`)

# --- NOT in apt, install by hand (see README.md) ---
#   herdr            -> github.com/herdrdev/herdr releases, binary into ~/.local/bin
#   yazi + ya        -> github.com/sxyazi/yazi releases (x86_64-unknown-linux-gnu.zip) into ~/.local/bin
#   Iosevka Nerd Font -> ~/.local/share/fonts/iosevka_nerd_font.ttf  (waybar/wofi/kitty)
#   vim-plug         -> curl into ~/.config/nvim/autoload/plug.vim
